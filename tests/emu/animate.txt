# Each frame for thirty ticks, then the next, and after four the first again.
# The timer has been running since the first game began, and is never reset,
# so its phase here depends on how many ticks the walk took. Waiting for the
# first frame to arrive and then going fifteen frames on puts every peek below
# in the middle of a step rather than on its edge.
let water = XRAM_PF+1*10+4
let frame0 = XRAM_ANIM+20*288
wait xram:water frame0&255 130
run 15
peek xram:water frame0&255 frame0>>8
run 30
peek xram:water (frame0+288)&255 (frame0+288)>>8
run 30
peek xram:water (frame0+2*288)&255 (frame0+2*288)>>8
run 30
peek xram:water (frame0+3*288)&255 (frame0+3*288)>>8
run 30
peek xram:water frame0&255 frame0>>8
